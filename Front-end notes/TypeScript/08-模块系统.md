# TypeScript 模块系统

## 1. 导出

### 1.1 导出声明

```typescript
// 导出变量
export const PI = 3.14;
export const E = 2.71;

// 导出函数
export function add(a: number, b: number): number {
  return a + b;
}

// 导出接口
export interface Point {
  x: number;
  y: number;
}

// 导出类型别名
export type Color = "red" | "green" | "blue";
```

### 1.2 导出多个

```typescript
function max<T>(arr: T[]): T {
  return arr.reduce((a, b) => (a > b ? a : b));
}

function min<T>(arr: T[]): T {
  return arr.reduce((a, b) => (a < b ? a : b));
}

export { max, min };
```

### 1.3 重命名导出

```typescript
interface A {
  x: number;
}

interface B {
  y: string;
}

export { A as TypeA, B as TypeB };
```

### 1.4 默认导出

```typescript
// 默认导出（每个文件只能有一个）
export default class Calculator {
  add(a: number, b: number): number {
    return a + b;
  }
}

// 或者
const calculator = new Calculator();
export default calculator;
```

## 2. 导入

### 2.1 基本导入

```typescript
import { Point } from "./types";
import { add } from "./math";

let p: Point = { x: 0, y: 0 };
let sum = add(1, 2);
```

### 2.2 重命名导入

```typescript
import { Point as P } from "./types";

let p: P = { x: 0, y: 0 };
```

### 2.3 导入模块全体

```typescript
import * as MathUtils from "./math";

MathUtils.add(1, 2);
MathUtils.max([1, 2, 3]);
```

### 2.4 默认导入

```typescript
import Calculator from "./Calculator";

let calc = new Calculator();
```

### 2.5 副作用导入

```typescript
// 只执行模块，不导入任何绑定
import "./polyfills";
```

## 3. 重新导出

```typescript
// 从其他模块导出
export { Point } from "./types";
export { add, multiply } from "./math";

// 重新导出并重命名
export { Point as GeoPoint } from "./types";

// 整个模块重导出
export * from "./types";
export * from "./math";
```

## 4. 模块类型

### 4.1 esModuleInterop

```json
{
  "compilerOptions": {
    "esModuleInterop": true
  }
}
```

```typescript
import fs from "fs";  // 不需要 require('fs')
```

### 4.2 AMD / CommonJS / UMD / System / ES2015

```typescript
// CommonJS 导出
export = {
  name: "MyModule",
  value: 42
};

// 导入 CommonJS 模块
import myModule = require("./myModule");
```

## 5. 动态导入

```typescript
// 懒加载模块
async function loadModule() {
  const { add } = await import("./math");
  return add(1, 2);
}

// 条件导入
async function loadFeature(flag: boolean) {
  if (flag) {
    const { FeatureA } = await import("./features/featureA");
    return new FeatureA();
  } else {
    const { FeatureB } = await import("./features/featureB");
    return new FeatureB();
  }
}
```

## 6. 模块声明

### 6.1 声明全局模块

```typescript
// globals.d.ts
declare module "my-library" {
  export function doSomething(): void;
  export const VERSION: string;
}

// 使用
import { doSomething } from "my-library";
doSomething();
```

### 6.2 声明模块扩展

```typescript
// express扩展
declare module "express" {
  interface Application {
    foo(): void;
  }
}
```

## 7. namespace（已废弃）

```typescript
// Math.ts
namespace Math {
  export function add(a: number, b: number): number {
    return a + b;
  }
}

// 使用
/// <reference path="Math.ts" />
Math.add(1, 2);
```

推荐使用 ES 模块替代。
