## 6. 混沌测试补充

基于实际组件故障场景：

| 场景 | 模拟方法 | 预期行为 |
|------|---------|---------|
| Inngest 不可用 | 停止 Inngest Dev Server | 项目卡在 queued，定时任务修复 |
| PostgreSQL 连接断开 | `pg_ctl stop` | 所有写操作失败，不丢失数据 |
| R2 上传超时 | 网络延迟注入(5s+) | 重试 3 次后标记失败 |
| Worker OOM | `docker run --memory=512M` | Inngest 重试，不阻塞后续任务 |

```ts
// src/server/services/cost-guard.service.ts
export async function checkCostGuard(userId: string) {
  const todayTokens = await prisma.usageRecord.aggregate({
    _sum: { quantity: true },
    where: {
      userId,
      resourceType: 'ai_tokens',
      recordedAt: { gte: todayStart }
    }
  });

  const COST_THRESHOLD = 100_000; // 10 万 tokens/天
  if ((todayTokens._sum.quantity || 0) > COST_THRESHOLD) {
    // 发送告警
    console.error(`User ${userId} exceeded token threshold`);
  }
}
```
